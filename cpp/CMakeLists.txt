cmake_minimum_required(VERSION 3.21)
project(MarkdownNative VERSION 0.261115 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets Core5Compat Test)
add_library(markdown_core STATIC document.cpp rendering.cpp resources.qrc)
target_link_libraries(markdown_core PUBLIC Qt6::Widgets Qt6::Core5Compat)
target_include_directories(markdown_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(markdown_ui STATIC editor.cpp)
target_link_libraries(markdown_ui PUBLIC markdown_core)
qt_add_executable(MARKDOWN WIN32 main.cpp)
target_link_libraries(MARKDOWN PRIVATE markdown_ui)
if(MSVC)
  add_compile_options(/utf-8)
  target_compile_options(markdown_core PRIVATE /utf-8)
  target_compile_options(markdown_ui PRIVATE /utf-8)
  target_compile_options(MARKDOWN PRIVATE /utf-8)
endif()
enable_testing()
qt_add_executable(markdown_tests tests.cpp)
target_link_libraries(markdown_tests PRIVATE markdown_ui Qt6::Test)
add_test(NAME markdown_tests COMMAND markdown_tests -platform offscreen -o test-results.txt,txt)
